package com.example.serialpaste

import android.app.Service
import android.content.Intent
import android.os.IBinder
import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.content.ClipboardManager
import android.content.ClipData
import android.content.Context
import java.io.InputStream
import java.util.UUID
import android.bluetooth.BluetoothSocket

class SerialService : Service() {

    private var running = true
    private var socket: BluetoothSocket? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        val addr = intent?.getStringExtra("address") ?: return START_NOT_STICKY
        startForeground(1, buildNotification())

        Thread {
            while (running) {
                try {
                    socket = BluetoothHelper.connect(addr)
                    val input: InputStream = socket!!.inputStream
                    val buf = ByteArray(128)

                    while (running) {
                        val len = input.read(buf)
                        if (len > 0) {
                            val data = String(buf, 0, len)
                            LogManager.log("接收：$data")

                            copy(data)
                            paste()
                        }
                    }
                } catch (e: Exception) {
                    LogManager.log("异常：${e.message}")
                    Thread.sleep(3000)
                }
            }
        }.start()

        return START_STICKY
    }

    private fun copy(text: String) {
        val cm = getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
        cm.setPrimaryClip(ClipData.newPlainText("serial", text))
    }

    private fun paste() {
        Intent(this, PasteAccessibilityService::class.java).also {
            it.action = "PASTE_ONCE"
            startService(it)
        }
    }

    private fun buildNotification(): Notification {
        val channel = NotificationChannel(
            "serial",
            "串口服务",
            NotificationManager.IMPORTANCE_LOW
        )
        getSystemService(NotificationManager::class.java).createNotificationChannel(channel)

        return Notification.Builder(this, "serial")
            .setContentTitle("串口服务运行中")
            .build()
    }

    override fun onDestroy() {
        running = false
        socket?.close()
    }

    override fun onBind(intent: Intent?): IBinder? = null
}
